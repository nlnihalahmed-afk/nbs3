    // Tab Switching Functionality
    function switchTab(tabId) {
      const sections = document.querySelectorAll('.page-section');
      sections.forEach(sec => sec.classList.remove('active'));

      const navBtns = document.querySelectorAll('.nav-btn');
      navBtns.forEach(btn => {
        btn.classList.remove('bg-sage-700', 'text-white', 'shadow-sm', 'font-bold');
        btn.classList.add('text-sage-800');
      });

      const activeSection = document.getElementById('sec-' + tabId);
      if (activeSection) {
        activeSection.classList.add('active');
      }

      const activeNavBtn = document.getElementById('nav-' + tabId);
      if (activeNavBtn) {
        activeNavBtn.classList.add('bg-sage-700', 'text-white', 'shadow-sm', 'font-bold');
        activeNavBtn.classList.remove('text-sage-800');
      }

      window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    // Mobile Navigation Drawer Toggle
    function toggleMobileMenu() {
      const menu = document.getElementById('mobile-menu');
      const icon = document.getElementById('menu-icon');
      if (menu.classList.contains('hidden')) {
        menu.classList.remove('hidden');
        icon.classList.remove('fa-bars');
        icon.classList.add('fa-xmark');
      } else {
        menu.classList.add('hidden');
        icon.classList.remove('fa-xmark');
        icon.classList.add('fa-bars');
      }
    }

    // Modal Dynamic Populating & Controls
    function openProfileModal(memberId) {
      const data = membersData[memberId];
      if (!data) return;

      document.getElementById('m-photo').src = data.photo;
      document.getElementById('m-role').textContent = data.role;
      document.getElementById('m-name').textContent = data.name;
      document.getElementById('m-tag').textContent = data.tag;
      document.getElementById('m-nickname').textContent = data.nickname;
      document.getElementById('m-hobby').textContent = data.hobby;
      document.getElementById('m-bio').textContent = data.bio;
      document.getElementById('m-school').textContent = data.school;
      document.getElementById('m-college').textContent = data.college;

      document.getElementById('m-fb').href = data.fb;
      document.getElementById('m-insta').href = data.insta;
      document.getElementById('m-wa').href = data.whatsapp;
      document.getElementById('m-tg').href = data.telegram;

      switchModalSubTab('personal');

      const modal = document.getElementById('profile-modal');
      modal.classList.remove('hidden');
      document.body.style.overflow = 'hidden';
    }

    function closeProfileModal() {
      const modal = document.getElementById('profile-modal');
      modal.classList.add('hidden');
      document.body.style.overflow = 'auto';
    }

    function switchModalSubTab(subTab) {
      const subContents = document.querySelectorAll('.modal-subcontent');
      subContents.forEach(c => c.classList.add('hidden'));

      ['personal', 'academic', 'social'].forEach(t => {
        const btn = document.getElementById('mtab-' + t);
        if (btn) {
          btn.classList.remove('text-sage-800', 'border-sage-700', 'font-bold');
          btn.classList.add('text-sage-600', 'border-transparent');
        }
      });

      const activeContent = document.getElementById('mcontent-' + subTab);
      if (activeContent) activeContent.classList.remove('hidden');

      const activeTabBtn = document.getElementById('mtab-' + subTab);
      if (activeTabBtn) {
        activeTabBtn.classList.add('text-sage-800', 'border-sage-700', 'font-bold');
        activeTabBtn.classList.remove('text-sage-600', 'border-transparent');
      }
    }

    document.addEventListener('keydown', function(e) {

      }
